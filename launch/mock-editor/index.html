// Draws a square grid onto a 2D canvas.

export interface GridOptions {
  step: number;
  majorEvery: number;
  minorColor: string;
  majorColor: string;
}

const defaults: GridOptions = {
  step: 24,
  majorEvery: 5,
  minorColor: '#2c2d31',
  majorColor: '#3a3b40',
};

function line(ctx: CanvasRenderingContext2D, x0: number, y0: number, x1: number, y1: number): void {
  ctx.beginPath();
  ctx.moveTo(x0, y0);
  ctx.lineTo(x1, y1);
  ctx.stroke();
}

export function drawGrid(ctx: CanvasRenderingContext2D, width: number, height: number, options: Partial<GridOptions> = {}): void {
  const o = { ...defaults, ...options };
  ctx.lineWidth = 1;
  for (let i = 0, x = 0; x <= width; x += o.step, i++) {
    ctx.strokeStyle = i % o.majorEvery === 0 ? o.majorColor : o.minorColor;
    line(ctx, x + 0.5, 0, x + 0.5, height);
  }
  for (let j = 0, y = 0; y <= height; y += o.step, j++) {
    ctx.strokeStyle = j % o.majorEvery === 0 ? o.majorColor : o.minorColor;
    line(ctx, 0, y + 0.5, width, y + 0.5);
  }
}

// Number of whole cells that fit along one axis.
export const cellsAcross = (length: number, step: number): number => Math.floor(length / step);
